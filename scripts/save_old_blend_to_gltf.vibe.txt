cat > /tmp/blend_inspect_export.py << 'EOF'
import bpy, mathutils, sys, os

# Get the output path from environment
out_glb = os.environ.get('OUT_GLB')

# Export to GLB
bpy.ops.export_scene.gltf(
    filepath=out_glb,
    export_format='GLB',
    export_apply=True,
    export_texcoords=True,
    export_normals=True,
    export_materials='EXPORT',
)

# Measure bounding box of all mesh objects in world space
all_coords = []
for obj in bpy.data.objects:
    if obj.type == 'MESH':
        for corner in obj.bound_box:
            wv = obj.matrix_world @ mathutils.Vector(corner)
            all_coords.append(wv)

if all_coords:
    xs = [c.x for c in all_coords]
    ys = [c.y for c in all_coords]
    zs = [c.z for c in all_coords]
    print(f"DIMS_BL: W={max(xs)-min(xs):.4f} D={max(ys)-min(ys):.4f} H={max(zs)-min(zs):.4f}")
    print(f"MIN_BL: X={min(xs):.4f} Y={min(ys):.4f} Z={min(zs):.4f}")
    print(f"MAX_BL: X={max(xs):.4f} Y={max(ys):.4f} Z={max(zs):.4f}")
    cx = (min(xs)+max(xs))/2
    cy = (min(ys)+max(ys))/2
    print(f"CENTER_XY_BL: {cx:.4f} {cy:.4f}")
    print(f"BOTTOM_Z_BL: {min(zs):.4f}")
EOF
echo "Script written"

OUT_GLB="$TEMPDIR/armchair/armchair.glb" blender --background \
  "$TEMPDIR/armchair/3D_armchair_1-yRj.blend" \
  --python /tmp/blend_inspect_export.py 2>&1 | grep -E "DIMS_BL|MIN_BL|MAX_BL|CENTER|BOTTOM|EXPORT_DONE"
